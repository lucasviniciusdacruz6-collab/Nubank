        function toggleBalanceVisibility() {
            appState.user.hideBalance = !appState.user.hideBalance;
            renderApp();
        }

        function toggleTheme() {
            const html = document.documentElement;
            if(html.classList.contains('dark')) {
                html.classList.remove('dark');
                document.getElementById('theme-icon').className = "fa-solid fa-sun text-sm";
            } else {
                html.classList.add('dark');
                document.getElementById('theme-icon').className = "fa-solid fa-moon text-sm";
            }
        }

        function toggleCardBlock() {
            appState.user.cardBlocked = !appState.user.cardBlocked;
            showToast(appState.user.cardBlocked ? 'Cartão bloqueado com sucesso.' : 'Cartão desbloqueado.');
        }

        function simulatedPixAction(type) {
            const amount = type === 'send' ? 150.00 : 500.00;
            if(type === 'send' && appState.user.balance < amount) {
                showToast('Saldo insuficiente para simulação!');
                return;
            }
            const oldVal = appState.user.balance;
            appState.user.balance = type === 'send' ? appState.user.balance - amount : appState.user.balance + amount;
            
            showToast(type === 'send' ? 'Pix enviado com sucesso! (Simulação)' : 'Pix recebido com sucesso! (Simulação)');
            renderApp();
        }

        function adminModifyBalance(action) {
            const inputVal = parseFloat(document.getElementById('admin-amount-input').value);
            if(isNaN(inputVal) || inputVal <= 0) {
                alert('Insira um valor válido.');
                return;
            }

            const oldValFormatted = `R$ ${appState.user.balance.toFixed(2)}`;
            if(action === 'add') appState.user.balance += inputVal;
            else if(action === 'remove') appState.user.balance -= inputVal;
            else if(action === 'set') appState.user.balance = inputVal;

            const newValFormatted = `R$ ${appState.user.balance.toFixed(2)}`;

            // Registrar Log
            const now = new Date();
            const dateStr = `${String(now.getDate()).padStart(2,'0')}/${String(now.getMonth()+1).padStart(2,'0')}/${now.getFullYear()} — ${String(now.getHours()).padStart(2,'0')}:${String(now.getMinutes()).padStart(2,'0')}`;
            
            appState.adminLogs.unshift({
                date: dateStr,
                user: appState.user.name,
                op: 'Alteração de saldo fictício',
                oldVal: oldValFormatted,
                newVal: newValFormatted,
                admin: 'Administrador Master'
            });

            const feedback = document.getElementById('admin-feedback');
            feedback.innerText = "Saldo de demonstração atualizado.";
            feedback.classList.remove('hidden');
            setTimeout(() => feedback.classList.add('hidden'), 3000);
            renderApp();
        }

        function showToast(msg) {
            const toast = document.getElementById('toast');
            toast.innerText = msg;
            toast.classList.remove('hidden');
            setTimeout(() => toast.classList.add('hidden'), 3000);
        }

        function openModal(id) {
            document.getElementById(id).classList.remove('hidden');
            if(id === 'notifications-modal') {
                document.getElementById('notifications-list').innerHTML = appState.notifications.map(n => `

                `).join('');
            }
        }
